import requests

# 🔗 Your ngrok URL

URL = "https://diminish-overcook-venus.ngrok-free.dev/api/v2/execute"

# 🧠 Language → version mapping

VERSIONS = {

    "python": "3.12.0",

    "java": "15.0.2",

    "cpp": "10.2.0"

}

def run_code(language, code, stdin=""):

    payload = {

        "language": language,

        "version": VERSIONS.get(language, "*"),

        "files": [

            {

                "content": code

            }

        ],

        "stdin": stdin

    }

    response = requests.post(URL, json=payload)

    data = response.json()

    run = data.get("run", {})

    compile = data.get("compile", {})

    # 🔥 Handle compile errors

    if compile.get("stderr"):

        return {

            "status": "COMPILE_ERROR",

            "error": compile["stderr"]

        }

    # 🔥 Handle runtime errors

    if run.get("stderr"):

        return {

            "status": "RUNTIME_ERROR",

            "error": run["stderr"]

        }

    return {

        "status": "SUCCESS",

        "output": run.get("stdout", ""),

        "exit_code": run.get("code", 0)

    }

# =========================

# 🚀 TEST CASES

# =========================

test_cases = [

    ("5\n1 2 3 4 5", "15"),

    ("3\n1000000000 1000000000 1000000000", "3000000000"),

    ("6\n-10 20 -30 40 -50 60", "30")

]

# =========================

# 🔥 CODE SNIPPETS

# =========================

codes = {

    "python": "n=int(input()); arr=list(map(int,input().split())); print(sum(arr))",

    "java": """import java.util.*;

public class Main {

    public static void main(String[] args){

        Scanner sc=new Scanner(System.in);

        int n=sc.nextInt();

        long sum=0;

        for(int i=0;i<n;i++) sum+=sc.nextLong();

        System.out.println(sum);

    }

}""",

    "cpp": """#include <iostream>

using namespace std;

int main(){

    int n; cin>>n;

    long long sum=0,x;

    for(int i=0;i<n;i++){

        cin>>x;

        sum+=x;

    }

    cout<<sum;

}"""

}

# =========================

# ▶️ RUN ALL TESTS

# =========================

for lang in ["python", "java", "cpp"]:

    print(f"\n=== Testing {lang.upper()} ===")

    for i, (inp, expected) in enumerate(test_cases, 1):

        result = run_code(lang, codes[lang], inp)

        if result["status"] != "SUCCESS":

            print(f"Test {i}: ❌ {result['status']}")

            print(result.get("error", ""))

            continue

        output = result["output"].strip()

        if output == expected:

            print(f"Test {i}: ✅ PASS")

            print("Expected:", expected)

            print("Got:", output)

        else:

            print(f"Test {i}: ❌ FAIL")
